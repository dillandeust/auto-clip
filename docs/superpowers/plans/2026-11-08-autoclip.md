# AutoClip Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ajouter à un fork d'OpenShorts un module de publication Playwright gratuit (TikTok/Instagram/YouTube) piloté par une file SQLite et un worker de planification, avec repli manuel, le tout en gardant le dashboard OpenShorts quasi inchangé.

**Architecture:** OpenShorts (cloné dans `_ref_openshorts/`) fournit ingestion → transcription → détection de moments (via Ollama) → rendu + dashboard de review. On ajoute un package Python `autoclip/` : une file `post_queue` en SQLite, un worker standalone qui poste les clips dus via des adaptateurs Playwright par plateforme, et on remplace le corps de `post_to_socials()` (`_ref_openshorts/app.py:6319`) pour enfiler dans cette file au lieu d'appeler Upload-Post. Le repli manuel est servi par une petite page HTML autonome montée sur FastAPI.

**Tech Stack:** Python 3.11, Playwright (sync API, Chromium persistant), SQLite (WAL), pytest, FastAPI (existant, OpenShorts), Ollama (OpenAI-compatible, local).

```python
# autoclip/queue.py
from __future__ import annotations
import sqlite3
from typing import Optional

PENDING = "pending"
IN_PROGRESS = "in_progress"
POSTED = "posted"
FAILED = "failed"
NEEDS_MANUAL = "needs_manual"

def init_db(conn: sqlite3.Connection) -> None:
    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS post_queue (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            job_id TEXT NOT NULL,
            clip_index INTEGER NOT NULL,
            clip_path TEXT NOT NULL,
            platform TEXT NOT NULL,
            profile TEXT NOT NULL DEFAULT 'default',
            caption TEXT NOT NULL DEFAULT '',
            scheduled_at TEXT,
            status TEXT NOT NULL DEFAULT 'pending',
            attempts INTEGER NOT NULL DEFAULT 0,
            last_error TEXT,
            posted_url TEXT,
            created_at TEXT NOT NULL,
            updated_at TEXT NOT NULL
        )
        """
    )

def enqueue(conn, *, job_id, clip_index, clip_path, platform, caption,
            scheduled_at: Optional[str], now: str, profile: str = "default") -> int:
    cur = conn.execute(
        """INSERT INTO post_queue
           (job_id, clip_index, clip_path, platform, profile, caption,
            scheduled_at, status, attempts, created_at, updated_at)
           VALUES (?,?,?,?,?,?,?,?,0,?,?)""",
        (job_id, clip_index, clip_path, platform, profile, caption,
         scheduled_at, PENDING, now, now),
    )
    return int(cur.lastrowid)

def get(conn, rid: int):
    return conn.execute("SELECT * FROM post_queue WHERE id=?", (rid,)).fetchone()

def due(conn, *, now: str):
    return conn.execute(
        """SELECT * FROM post_queue
           WHERE status=? AND (scheduled_at IS NULL OR scheduled_at <= ?)
           ORDER BY id ASC""",
        (PENDING, now),
    ).fetchall()

def list_by_status(conn, status: str):
    return conn.execute(
        "SELECT * FROM post_queue WHERE status=? ORDER BY id ASC", (status,)
    ).fetchall()

def mark_in_progress(conn, rid: int, *, now: str) -> None:
    conn.execute(
        "UPDATE post_queue SET status=?, updated_at=? WHERE id=?",
        (IN_PROGRESS, now, rid),
    )

def mark_posted(conn, rid: int, *, url: str, now: str) -> None:
    conn.execute(
        "UPDATE post_queue SET status=?, posted_url=?, updated_at=? WHERE id=?",
        (POSTED, url, now, rid),
    )

def mark_failed(conn, rid: int, *, error: str, max_attempts: int, now: str) -> None:
    row = get(conn, rid)
    attempts = (row["attempts"] or 0) + 1
    status = NEEDS_MANUAL if attempts >= max_attempts else PENDING
    conn.execute(
        "UPDATE post_queue SET status=?, attempts=?, last_error=?, updated_at=? WHERE id=?",
        (status, attempts, error, now, rid),
    )

def mark_needs_manual(conn, rid: int, *, error: str, now: str) -> None:
    conn.execute(
        "UPDATE post_queue SET status=?, last_error=?, updated_at=? WHERE id=?",
        (NEEDS_MANUAL, error, now, rid),
    )

def cancel(conn, rid: int) -> None:
    conn.execute("DELETE FROM post_queue WHERE id=? AND status=?", (rid, PENDING))
```

```python
# autoclip/poster/base.py
from __future__ import annotations
from dataclasses import dataclass
from pathlib import Path
from typing import Optional, Protocol

@dataclass
class PostResult:
    ok: bool
    url: Optional[str] = None
    error: Optional[str] = None
    screenshot: Optional[str] = None

class LoginRequired(Exception):
    """Levée quand la session navigateur n'est plus connectée."""

class Poster(Protocol):
    name: str
    def post(self, clip_path: str, caption: str, *,
             profile: str = "default", dry_run: bool = False) -> PostResult: ...

def session_dir(platform: str, profile: str, *, profiles_root: str = "profiles") -> Path:
    d = Path(profiles_root) / platform / profile
    d.mkdir(parents=True, exist_ok=True)
    return d

def screenshot_path(platform: str, label: str, *, logs_root: str = "logs") -> str:
    Path(logs_root).mkdir(parents=True, exist_ok=True)
    return str(Path(logs_root) / f"{platform}_{label}.png")
```

```python
# autoclip/worker.py
from __future__ import annotations
import time
from datetime import datetime, timezone
from typing import Dict

from . import queue
from .db import connect
from .config import load
from .poster.base import Poster, PostResult, LoginRequired

def _now() -> str:
    return datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")

def run_once(conn, posters: Dict[str, Poster], *, now: str, max_attempts: int) -> int:
    """Traite tous les posts dus. Retourne le nombre traité."""
    rows = queue.due(conn, now=now)
    for row in rows:
        rid = row["id"]
        platform = row["platform"]
        queue.mark_in_progress(conn, rid, now=now)
        poster = posters.get(platform)
        if poster is None:
            queue.mark_needs_manual(
                conn, rid, error=f"aucun adaptateur pour '{platform}'", now=now)
            continue
        try:
            result: PostResult = poster.post(
                row["clip_path"], row["caption"], profile=row["profile"])
            if result.ok:
                queue.mark_posted(conn, rid, url=result.url or "", now=now)
            else:
                queue.mark_failed(conn, rid, error=result.error or "échec",
                                  max_attempts=max_attempts, now=now)
        except LoginRequired as e:
            queue.mark_needs_manual(conn, rid, error=f"login requis: {e}", now=now)
        except Exception as e:  # garde-fou : tout autre échec -> retry/needs_manual
            queue.mark_failed(conn, rid, error=str(e),
                              max_attempts=max_attempts, now=now)
    return len(rows)

def run_forever(poll_seconds: int = 15) -> None:
    from .poster.tiktok import TikTokPoster
    from .poster.instagram import InstagramPoster
    from .poster.youtube import YouTubePoster
    cfg = load()
    conn = connect(cfg.db_path)
    queue.init_db(conn)
    posters = {
        "tiktok": TikTokPoster(cfg),
        "instagram": InstagramPoster(cfg),
        "youtube": YouTubePoster(cfg),
    }
    print(f"[autoclip] worker démarré (poll={poll_seconds}s, db={cfg.db_path})")
    while True:
        n = run_once(conn, posters, now=_now(), max_attempts=cfg.max_attempts)
        if n:
            print(f"[autoclip] {n} post(s) traité(s)")
        time.sleep(poll_seconds)
```

```python
        page.goto(LOGIN_URLS[platform])
        print(f"\n[autoclip] Connecte-toi à {platform} dans la fenêtre, "
              f"puis reviens ici et appuie sur Entrée pour sauvegarder la session.")

        ctx.close()  # la session persiste dans user_data_dir
```

```markdown
# AutoClip

Clipping automatique (Twitch/Kick/YouTube/fichiers) + publication semi-auto
(TikTok/Instagram/YouTube) via Playwright, basé sur un fork d'OpenShorts.

## Installation
1. `pip install -e ".[dev]" && python -m playwright install chromium`
2. Installer ffmpeg + pilotes NVIDIA/CUDA.
3. Ollama : `ollama serve` puis `ollama pull llama3.1:8b`.
4. Copier `.env.example` → `.env` (et configurer `_ref_openshorts/.env`).

## Connexion aux plateformes (une fois)
```
autoclip login tiktok
autoclip login instagram
autoclip login youtube
```

## Usage quotidien
1. Démarrer le backend + dashboard OpenShorts (voir `_ref_openshorts/`).
2. Démarrer le worker : `autoclip worker`.
3. Dans le dashboard : coller une URL / un fichier → valider les clips → planifier.
4. Les posts dus sont publiés par le worker. Les échecs → `http://localhost:8000/manual`.

## Limites
- Automatisation navigateur = contre les CGU des plateformes (risque de ban ; petit volume conseillé).
- Les sélecteurs Playwright peuvent nécessiter un ajustement si une plateforme change son UI
  (les captures d'erreur sont dans `logs/`).
- Droits sur le contenu source : responsabilité de l'utilisateur.
```
